import os, re, urllib.parse, httpx, yt_dlp
from bs4 import BeautifulSoup
from telegram import Update, WebAppInfo, InlineKeyboardButton, InlineKeyboardMarkup
from telegram.ext import Application, CommandHandler, MessageHandler, ContextTypes, filters

BOT_TOKEN = os.getenv("BOT_TOKEN", "PUT_YOUR_BOT_TOKEN_HERE")
WEBAPP_URL = os.getenv("WEBAPP_URL", "https://your-site.github.io/teletube_player.html")

def player_url(src="", title="", channel="", scan=""):
    q = {k: v for k, v in {"src": src, "title": title, "channel": channel, "scan": scan}.items() if v}
    return f"{WEBAPP_URL}?{urllib.parse.urlencode(q)}" if q else WEBAPP_URL

def discover_site_content(url: str, max_items: int = 10):
    domain = urllib.parse.urlparse(url).netloc.replace("www.", "")
    items = []
    # 1) فحص عبر yt-dlp (يدعم Crunchyroll وYouTube وTwitch و1800+ موقع)
    try:
        with yt_dlp.YoutubeDL({"extract_flat": "in_playlist", "quiet": True, "playlistend": max_items}) as ydl:
            info = ydl.extract_info(url, download=False)
            for e in (info.get("entries") or [info])[:max_items]:
                if e and (e.get("url") or e.get("webpage_url")):
                    items.append({"title": e.get("title", "حلقة جديدة"), "url": e.get("url") or e.get("webpage_url")})
    except Exception:
        pass
    # 2) فحص HTML وبطاقات الحلقات عبر BeautifulSoup4
    if not items:
        r = httpx.get(url, follow_redirects=True, timeout=10)
        soup = BeautifulSoup(r.text, "html.parser")
        for a in soup.find_all("a", href=True):
            if re.search(r"(watch|episode|series|anime|video)", a["href"], re.I):
                items.append({"title": a.get_text(" ", strip=True)[:70] or "عرض مكتشف", "url": urllib.parse.urljoin(url, a["href"])})
                if len(items) >= max_items: break
    return domain, items

async def on_url(update: Update, context: ContextTypes.DEFAULT_TYPE):
    url = update.message.text.strip()
    domain, items = discover_site_content(url)
    btns = [[InlineKeyboardButton(f"🌐 فتح محتوى {domain} في المشغل", web_app=WebAppInfo(url=player_url(scan=url)))]]
    for ep in items[:6]:
        btns.append([InlineKeyboardButton(f"▶️ {ep['title'][:40]}", web_app=WebAppInfo(url=player_url(src=ep['url'], title=ep['title'], channel=domain)))])
    await update.message.reply_text(f"🎯 تم اكتشاف {len(items)} عرض/حلقة نازلة في {domain}:", reply_markup=InlineKeyboardMarkup(btns))

app = Application.builder().token(BOT_TOKEN).build()
app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, on_url))
app.run_polling()
